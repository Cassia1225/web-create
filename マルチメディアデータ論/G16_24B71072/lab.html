#include <stdio.h>

int main(void)
{
    int score;

    printf("点数を入力してください：");
    scanf("%d", &score);

        printf("合格です。\n");
    }else{
        printf("不合格です。\n");

このプログラムは、入力された点数が60点以上なら「合格」、60点未満なら「不合格」と表示する C言語のサンプルです。変数、条件分岐（if文）、標準入出力（scanf・printf）の基本的な使い方を 学ぶことができます。

using UnityEngine;

public class SimpleMover : MonoBehaviour
{
  // Inspector から編集できるフィールド
  public float speed = 2.0f;

  // 最初の回だけ呼ばれる 1
  void Start()
  {
    Debug.Log("SimpleMover started");
  }

  // 毎フレーム呼ばれる描画と同期、可変レート ()
  void Update()
  {
  // Time.deltaTime は前フレームからの経過時間秒[]
  transform.Translate(Vector3.right * speed * Time.deltaTime);
  }
}

このプログラムは、Unity上でオブジェクトを一定の速度で右方向に移動させるスクリプトです。Startメソッドは開始時に一度だけ実行され、Updateメソッドは毎フレーム実行されます。speedの値を変更することで移動速度を調整することができます。

見坐地研究室

見坐地研究室では、自動車の防音材の遮音特性や音響特性、そしてそれらが人体に与える影響について研究しています。 静粛性の高い快適な車内空間の実現を目指しています。

ゼミナールで学ぶ内容

遮音特性や音響の基礎理論を出発点として、微分方程式をはじめとする高等数学を用いながら、 波動方程式、SEA法、防音材のBiot理論、さらに機械学習の基礎と応用について学び、 研究を進めます。また、課題では出された内容に対して次の授業に発表を行い意見を交換し合い内容理解を深めています。

以下は、質量則(Mass Law)を用いて防音材の音響透過損失(遮音性能)を 計算するプログラムの例です。 

#include <stdio.h>
#include <math.h>

int main(void) {
    double surface_density = 8.0;  
    double freq = 500.0;           

    double TL = 20 * log10(surface_density * freq) - 47.4;

    printf("面密度: %.1f kg/m^2\n", surface_density);
    printf("周波数: %.1f Hz\n", freq);
    printf("透過損失 TL = %.2f dB\n", TL);

    return 0;
}
